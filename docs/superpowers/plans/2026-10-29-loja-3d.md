# Loja 3D — Plano de implementação

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Transformar a aba `/3d` numa loja — home com banner/seções, página própria por produto (galeria + vídeo + cores) e admin pra gerenciar a mídia — mantendo o fluxo de encomenda por WhatsApp.

**Architecture:** Next.js 14 App Router + Supabase. A view `vitrine` ganha `fotos/cores/video_url`; a página de produto é um Server Component lido por slug; o card próprio passa a abrir `/3d/[slug]` em vez de ir direto pro `/r`. Lógica pura (palette de cores, anexar cor na mensagem) com testes `node:test`; telas verificadas por `npm run build` + preview no browser.

**Tech Stack:** Next.js 14 (App Router, TS), Supabase (`@supabase/supabase-js`), Tailwind, `next/image`, runner de teste nativo do Node (`node --test` via tsx).

**Spec:** `docs/superpowers/specs/2026-10-29-loja-3d-design.md`

**Convenções do projeto:** respostas e commits em pt-BR; sem emoji no código de UI; mobile-first (390px); cores da marca `areia`/`tinta`/`pirraia`/`fumo`; `numeric` do Supabase volta como string — sempre `Number()`. Deploy é manual no EasyPanel (nada vai ao ar até o dono dar Deploy).

/** Palette fixo de cores de filamento. Produto guarda só as chaves em cores[]. */
export interface Cor {
  label: string;
  hex: string;
}

export const PALETTE: Record<string, Cor> = {
  vermelho: { label: "Vermelho", hex: "#DC2626" },
  azul: { label: "Azul", hex: "#2563EB" },
  verde: { label: "Verde", hex: "#16A34A" },
  amarelo: { label: "Amarelo", hex: "#F59E0B" },
  roxo: { label: "Roxo", hex: "#7C3AED" },
  laranja: { label: "Laranja", hex: "#EA580C" },
  preto: { label: "Preto", hex: "#111827" },
  rosa: { label: "Rosa", hex: "#EC4899" },
  branco: { label: "Branco", hex: "#F8FAFC" },
  cinza: { label: "Cinza", hex: "#6B7280" },
};

export function corLabel(chave: string): string {
  return PALETTE[chave]?.label ?? chave;
}

/** Normaliza as chaves de cor de um produto em objetos renderizáveis (ordem preservada). */
export function coresDoProduto(
  chaves: string[] | null | undefined,
): Array<{ chave: string; label: string; hex: string }> {
  if (!chaves) return [];
  return chaves
    .filter((c) => c in PALETTE)
    .map((c) => ({ chave: c, label: PALETTE[c].label, hex: PALETTE[c].hex }));
}
```

/**
 * Acrescenta " Cor: {label}." ao parâmetro text de uma URL wa.me.
 * Usado no /r/[slug] quando o visitante escolheu uma cor na página do produto.
 * No-op se não houver cor, se a URL não for wa.me, ou se já tiver "Cor:".
 */
export function anexarCor(shortUrl: string, corLabel: string | null | undefined): string {
  if (!corLabel) return shortUrl;
  let u: URL;
  try {
    u = new URL(shortUrl);
  } catch {
    return shortUrl;
  }
  if (!u.hostname.includes("wa.me")) return shortUrl;
  const texto = u.searchParams.get("text") ?? "";
  if (texto.includes("Cor:")) return shortUrl;
  u.searchParams.set("text", `${texto} Cor: ${corLabel}.`.trim());
  return u.toString();
}
```

import { supabasePublic } from "@/lib/supabase";
import type { VitrineItem } from "@/lib/types";

/** Produto próprio pela slug (só o que está na vitrine). null se não existir. */
export async function getProduto3D(slug: string): Promise<VitrineItem | null> {
  const supabase = supabasePublic();
  const { data } = await supabase
    .from("vitrine")
    .select("*")
    .eq("slug", slug)
    .eq("proprio", true)
    .maybeSingle<VitrineItem>();
  return data ?? null;
}

/** Até `limite` outras peças da mesma categoria (exclui o próprio slug). */
export async function relacionados3D(
  categoria: string | null,
  slugAtual: string,
  limite = 6,
): Promise<VitrineItem[]> {
  if (!categoria) return [];
  const supabase = supabasePublic();
  const { data } = await supabase
    .from("vitrine")
    .select("*")
    .eq("proprio", true)
    .eq("categoria", categoria)
    .neq("slug", slugAtual)
    .limit(limite)
    .returns<VitrineItem[]>();
  return data ?? [];
}
```

"use client";
import { useState } from "react";
import Image from "next/image";

type Midia = { tipo: "foto"; url: string } | { tipo: "video"; url: string };

export function Galeria3D({
  capa,
  fotos,
  videoUrl,
  titulo,
}: {
  capa: string | null;
  fotos: string[] | null | undefined;
  videoUrl: string | null | undefined;
  titulo: string;
}) {
  const midias: Midia[] = [];
  if (capa) midias.push({ tipo: "foto", url: capa });
  for (const f of fotos ?? []) if (f && f !== capa) midias.push({ tipo: "foto", url: f });
  if (videoUrl) midias.push({ tipo: "video", url: videoUrl });

  const [ativa, setAtiva] = useState(0);
  const atual = midias[ativa];

  if (!atual) {
    return <div className="aspect-square w-full rounded-2xl bg-black/5" />;
  }

  return (
    <div className="flex flex-col gap-3">
      <div className="relative aspect-square w-full overflow-hidden rounded-2xl bg-white shadow-carta">
        {atual.tipo === "foto" ? (
          <Image src={atual.url} alt={titulo} fill sizes="(max-width:1024px) 100vw, 50vw" className="object-cover" priority />
        ) : (
          <video src={atual.url} controls playsInline className="h-full w-full object-cover" />
        )}
      </div>
      {midias.length > 1 && (
        <div className="no-scrollbar flex gap-2 overflow-x-auto">
          {midias.map((m, i) => (
            <button
              key={m.url + i}
              type="button"
              onClick={() => setAtiva(i)}
              aria-label={m.tipo === "video" ? "Ver vídeo" : `Foto ${i + 1}`}
              className={`relative h-16 w-16 shrink-0 overflow-hidden rounded-lg border-2 ${
                i === ativa ? "border-pirraia" : "border-transparent"
              }`}
            >
              {m.tipo === "foto" ? (
                // eslint-disable-next-line @next/next/no-img-element
                <img src={m.url} alt="" className="h-full w-full object-cover" />
              ) : (
                <span className="flex h-full w-full items-center justify-center bg-tinta text-xs font-bold text-white">▶</span>
              )}
            </button>
          ))}
        </div>
      )}
    </div>
  );
}
```

// lib/banner.ts
import { supabasePublic } from "@/lib/supabase";
import type { Banner } from "@/lib/types";

export async function bannerAtivo(): Promise<Banner | null> {
  const supabase = supabasePublic();
  const { data } = await supabase
    .from("banners")
    .select("id, imagem_url, titulo, subtitulo, link")
    .eq("ativo", true)
    .order("ordem", { ascending: true })
    .limit(1)
    .maybeSingle<Banner>();
  return data ?? null;
}

```ts
export async function adicionarFoto3D(formData: FormData) {
  const produtoId = Number(formData.get("produtoId"));
  const foto = formData.get("foto");
  const supabase = supabaseAdmin();
  if (!produtoId || !foto || typeof foto === "string" || foto.size === 0) {
    redirect("/admin/3d?erro=1");
  }
  const f = foto as File;
  if (!f.type.startsWith("image/") || f.size > 8 * 1024 * 1024) redirect("/admin/3d?erro=1");
  const ext = (f.name.split(".").pop() || "jpg").toLowerCase().replace(/[^a-z0-9]/g, "") || "jpg";
  const caminho = `3d/galeria/${produtoId}-${Date.now()}.${ext}`;
  const bytes = Buffer.from(await f.arrayBuffer());
  const { error: upErr } = await supabase.storage.from("criativos").upload(caminho, bytes, { contentType: f.type, upsert: true });
  if (upErr) redirect("/admin/3d?erro=1");
  const url = supabase.storage.from("criativos").getPublicUrl(caminho).data.publicUrl;
  const { data: prod } = await supabase.from("produtos").select("fotos").eq("id", produtoId).single();
  const fotos = [...((prod?.fotos as string[]) ?? []), url];
  await supabase.from("produtos").update({ fotos }).eq("id", produtoId);
  revalidatePath("/admin/3d");
  revalidatePath("/3d");
  redirect("/admin/3d?ok=1");
}

export async function removerFoto3D(formData: FormData) {
  const produtoId = Number(formData.get("produtoId"));
  const url = String(formData.get("url") ?? "");
  const supabase = supabaseAdmin();
  const { data: prod } = await supabase.from("produtos").select("fotos").eq("id", produtoId).single();
  const fotos = ((prod?.fotos as string[]) ?? []).filter((u) => u !== url);
  await supabase.from("produtos").update({ fotos }).eq("id", produtoId);
  revalidatePath("/admin/3d");
  revalidatePath("/3d");
  redirect("/admin/3d?ok=1");
}

export async function salvarCores3D(formData: FormData) {
  const produtoId = Number(formData.get("produtoId"));
  const cores = formData.getAll("cores").map(String).filter(Boolean);
  await supabaseAdmin().from("produtos").update({ cores }).eq("id", produtoId);
  revalidatePath("/admin/3d");
  revalidatePath("/3d");
  redirect("/admin/3d?ok=1");
}

export async function definirCategoria3D(formData: FormData) {
  const produtoId = Number(formData.get("produtoId"));
  const categoria = String(formData.get("categoria") ?? "").trim() || "3D";
  await supabaseAdmin().from("produtos").update({ categoria }).eq("id", produtoId);
  revalidatePath("/admin/3d");
  revalidatePath("/3d");
  redirect("/admin/3d?ok=1");
}

export async function salvarBanner(formData: FormData) {
  const supabase = supabaseAdmin();
  const titulo = String(formData.get("titulo") ?? "").trim() || null;
  const subtitulo = String(formData.get("subtitulo") ?? "").trim() || null;
  const link = String(formData.get("link") ?? "").trim() || null;
  const foto = formData.get("foto");
  let imagemUrl = String(formData.get("imagem_url") ?? "").trim() || null;
  if (foto && typeof foto !== "string" && foto.size > 0) {
    const f = foto as File;
    const ext = (f.name.split(".").pop() || "jpg").toLowerCase().replace(/[^a-z0-9]/g, "") || "jpg";
    const caminho = `3d/banner/${Date.now()}.${ext}`;
    const { error } = await supabase.storage.from("criativos").upload(caminho, Buffer.from(await f.arrayBuffer()), { contentType: f.type, upsert: true });
    if (!error) imagemUrl = supabase.storage.from("criativos").getPublicUrl(caminho).data.publicUrl;
  }
  const { data: existente } = await supabase.from("banners").select("id").eq("ativo", true).order("ordem").limit(1).maybeSingle();
  if (existente) {
    await supabase.from("banners").update({ imagem_url: imagemUrl, titulo, subtitulo, link }).eq("id", existente.id);
  } else {
    await supabase.from("banners").insert({ imagem_url: imagemUrl, titulo, subtitulo, link, ativo: true });
  }
  revalidatePath("/admin/3d");
  revalidatePath("/3d");
  redirect("/admin/3d?ok=1");
}
```
